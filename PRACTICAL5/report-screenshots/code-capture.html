Part 1: Backend – Supabase Client (src/lib/supabase.js)

TikTok_Server/src/lib/supabase.js
const { createClient } = require('@supabase/supabase-js');

const supabaseUrl = process.env.SUPABASE_URL;
const supabaseKey = process.env.SUPABASE_SERVICE_KEY;

if (!supabaseUrl || !supabaseKey) {
  console.error('Missing Supabase credentials. Check your environment variables.');
}

const supabase = supabaseUrl && supabaseKey
  ? createClient(supabaseUrl, supabaseKey)
  : null;

module.exports = supabase;

const uploadFile = async (bucketName, filePath, fileData, contentType) => {
  const client = ensureSupabase();
  const { data, error } = await client.storage
    .from(bucketName)
    .upload(filePath, fileData, { cacheControl: '3600', upsert: false, contentType });
  if (error) throw error;
  return { data, fileUrl: getPublicUrl(bucketName, filePath) };
};

const createSignedUploadUrl = async (bucketName, filePath) => {
  const { data, error } = await client.storage
    .from(bucketName)
    .createSignedUploadUrl(filePath);
  if (error) throw error;
  return data;
};
